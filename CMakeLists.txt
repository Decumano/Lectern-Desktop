cmake_minimum_required(VERSION 3.25)

# The release version, and the single source of truth for it: the binary's
# version resource, the CPack artifact names, and the number the updater
# compares against a manifest all read PROJECT_VERSION.
#
# CI passes the git tag here (-DLECTERN_VERSION=0.13.0). Without that the
# hard-coded default would ship inside every tagged build, and an updater that
# believes it is 0.12.11 would re-offer 0.13.0 forever after installing it.
set(LECTERN_VERSION "0.12.11" CACHE STRING
    "Release version, normally the git tag without its leading v")

if(NOT LECTERN_VERSION MATCHES "^[0-9]+\.[0-9]+\.[0-9]+$")
  message(FATAL_ERROR
    "LECTERN_VERSION must be MAJOR.MINOR.PATCH, got '${LECTERN_VERSION}'")
endif()

project(lectern-desktop-cpp VERSION ${LECTERN_VERSION} LANGUAGES CXX)

list(APPEND CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)

# saucer's public headers use C++23 (std::move_only_function, deducing this).
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /Zc:__cplusplus /bigobj)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

find_package(saucer REQUIRED)
find_package(cpr CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(unofficial-sodium CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)

# ── Updater configuration ──
#
# Both are empty by default, which disables the updater: a build with no public
# key must never accept a manifest. Release builds pass the key generated by
# `lectern-updater-tool keygen`; the matching secret key lives only in CI.
set(LECTERN_UPDATE_PUBLIC_KEY "" CACHE STRING
    "Ed25519 public key (hex) that release manifests are verified against")
set(LECTERN_UPDATE_MANIFEST_URL "" CACHE STRING
    "URL of the signed update manifest (latest.json)")

configure_file(
  ${CMAKE_CURRENT_SOURCE_DIR}/src/version.h.in
  ${CMAKE_CURRENT_BINARY_DIR}/generated/version.h
  @ONLY
)

# Everything but main() lives in a library so the tests can reach the sync
# engine, the merge algorithm and the updater.
add_library(lectern_desktop_core STATIC
  src/assets.cpp
  src/util.cpp
  src/paths.cpp
  src/workfolder.cpp
  src/dialogs.cpp
  src/diff.cpp
  src/cloud.cpp
  src/updater.cpp
)

target_include_directories(lectern_desktop_core PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}/src
  ${CMAKE_CURRENT_BINARY_DIR}/generated
)

target_link_libraries(lectern_desktop_core PUBLIC
  cpr::cpr
  nlohmann_json::nlohmann_json
  unofficial-sodium::sodium
  spdlog::spdlog
)

if(WIN32)
  # IFileDialog for the folder picker, ShellExecute for `reveal`.
  target_link_libraries(lectern_desktop_core PUBLIC Ole32 Shell32 Shlwapi)
elseif(NOT APPLE)
  # GTK 4 is already in the process on Linux — WebKitGTK links it — so using
  # it for the dialogs costs nothing and gains everything: GtkFileDialog goes
  # through xdg-desktop-portal inside a Flatpak sandbox, where no external
  # dialog helper exists and the shell-out path simply cannot work.
  # Without GTK, dialogs fall back to zenity/kdialog (see src/dialogs.cpp).
  find_package(PkgConfig)
  if(PkgConfig_FOUND)
    pkg_check_modules(GTK4 IMPORTED_TARGET gtk4)
  endif()
  if(TARGET PkgConfig::GTK4)
    target_link_libraries(lectern_desktop_core PUBLIC PkgConfig::GTK4)
    target_compile_definitions(lectern_desktop_core PRIVATE
      LECTERN_USE_GTK_DIALOGS)
    message(STATUS "Dialogs: GTK 4 (portal-aware)")
  else()
    message(STATUS "Dialogs: zenity/kdialog fallback (gtk4 not found)")
  endif()
endif()

# Running from a build tree: the frontend submodule sits in the source dir.
# Installed builds ignore this and resolve relative to the executable — see
# src/assets.cpp.
set(LECTERN_FRONTEND_DIR "${CMAKE_CURRENT_SOURCE_DIR}/frontend"
    CACHE PATH "Directory holding index.html and the frontend assets")
target_compile_definitions(lectern_desktop_core PRIVATE
  LECTERN_DEFAULT_FRONTEND_DIR="${LECTERN_FRONTEND_DIR}")

# ── The app ──

# WIN32/MACOSX_BUNDLE keep the release build from opening a console window,
# which is what the Rust build's `windows_subsystem = "windows"` did.
option(LECTERN_EMBED_FRONTEND
       "Link the frontend into the executable for a single-file portable build"
       OFF)

add_executable(lectern WIN32 MACOSX_BUNDLE
  src/main.cpp
  src/bridge.cpp
  src/embedded.cpp
  src/window_manager.cpp
)
target_link_libraries(lectern PRIVATE lectern_desktop_core saucer::saucer)

if(WIN32)
  # Gives the .exe its icon and version resource in Explorer.
  configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/packaging/lectern.rc.in
    ${CMAKE_CURRENT_BINARY_DIR}/generated/lectern.rc
    @ONLY
  )
  target_sources(lectern PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated/lectern.rc)
endif()

if(MSVC)
  # A WIN32 executable needs WinMain; this keeps the portable `int main()`.
  set_target_properties(lectern PROPERTIES LINK_FLAGS "/ENTRY:mainCRTStartup")
endif()

if(APPLE)
  set_target_properties(lectern PROPERTIES
    OUTPUT_NAME "Lectern"
    MACOSX_BUNDLE_BUNDLE_NAME "Lectern"
    MACOSX_BUNDLE_GUI_IDENTIFIER "com.lectern.app"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_ICON_FILE "lectern.icns"
    MACOSX_BUNDLE_INFO_PLIST
      "${CMAKE_CURRENT_SOURCE_DIR}/packaging/Info.plist.in"
  )
  # Bundle resources have to be listed as sources to be copied in.
  target_sources(lectern PRIVATE packaging/lectern.icns)
  set_source_files_properties(packaging/lectern.icns PROPERTIES
    MACOSX_PACKAGE_LOCATION "Resources")
endif()

# ── Install ──
#
# Layout differs per platform because that is what each packager expects:
#   Windows  <root>/lectern.exe, <root>/frontend/, the runtime DLLs beside it
#   macOS    Lectern.app/Contents/{MacOS,Resources/frontend}
#   Linux    /usr/bin/lectern, /usr/share/lectern/frontend, desktop + icons
# src/assets.cpp knows all three.

if(APPLE)
  install(TARGETS lectern BUNDLE DESTINATION . RUNTIME DESTINATION bin)
  if(NOT LECTERN_EMBED_FRONTEND)
    install(DIRECTORY "${LECTERN_FRONTEND_DIR}/"
            DESTINATION "Lectern.app/Contents/Resources/frontend"
            PATTERN ".git*" EXCLUDE)
  endif()

elseif(WIN32)
  install(TARGETS lectern RUNTIME DESTINATION .)
  if(NOT LECTERN_EMBED_FRONTEND)
    install(DIRECTORY "${LECTERN_FRONTEND_DIR}/"
            DESTINATION "frontend"
            PATTERN ".git*" EXCLUDE)
  endif()

  # The app is not self-contained the way the Tauri binary was: cpr, libcurl,
  # libsodium, spdlog, fmt, zlib and WebView2Loader are all separate DLLs.
  # Resolving them from the built binary means a new transitive dependency
  # can't be forgotten.
  install(CODE "
    set(CMAKE_INSTALL_UE_DIRECTORIES \"${CMAKE_BINARY_DIR}\")
    file(GET_RUNTIME_DEPENDENCIES
      EXECUTABLES \"$<TARGET_FILE:lectern>\"
      RESOLVED_DEPENDENCIES_VAR _resolved
      UNRESOLVED_DEPENDENCIES_VAR _unresolved
      DIRECTORIES \"${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/bin\"
      PRE_EXCLUDE_REGEXES \"api-ms-\" \"ext-ms-\"
      POST_EXCLUDE_REGEXES \".*[Ss]ystem32.*\"
    )
    foreach(_lib \${_resolved})
      file(INSTALL DESTINATION \"\${CMAKE_INSTALL_PREFIX}\"
           TYPE SHARED_LIBRARY FOLLOW_SYMLINK_CHAIN FILES \"\${_lib}\")
    endforeach()
    if(_unresolved)
      message(WARNING \"unresolved runtime dependencies: \${_unresolved}\")
    endif()
  ")

else()
  install(TARGETS lectern RUNTIME DESTINATION bin)
  if(NOT LECTERN_EMBED_FRONTEND)
    install(DIRECTORY "${LECTERN_FRONTEND_DIR}/"
            DESTINATION "share/lectern/frontend"
            PATTERN ".git*" EXCLUDE)
  endif()

  configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/packaging/lectern.desktop.in
    ${CMAKE_CURRENT_BINARY_DIR}/generated/com.lectern.app.desktop
    @ONLY
  )
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/generated/com.lectern.app.desktop
          DESTINATION share/applications)
  install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/packaging/com.lectern.app.metainfo.xml
          DESTINATION share/metainfo)

  install(FILES packaging/lectern-32.png
          DESTINATION share/icons/hicolor/32x32/apps RENAME com.lectern.app.png)
  install(FILES packaging/lectern-128.png
          DESTINATION share/icons/hicolor/128x128/apps RENAME com.lectern.app.png)
  install(FILES packaging/lectern-256.png
          DESTINATION share/icons/hicolor/256x256/apps RENAME com.lectern.app.png)
endif()

include(cmake/Packaging.cmake)
